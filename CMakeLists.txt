cmake_minimum_required(VERSION 3.25)
project(VaginaPlugin VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "Use the same static runtime in SDK and plugin" FORCE)
set(VST3_SDK_ROOT "" CACHE PATH "Official VST3 SDK 3.8.1 checkout")
if(NOT EXISTS "${VST3_SDK_ROOT}/CMakeLists.txt")
  include(FetchContent)
  FetchContent_Declare(vst3sdk GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git GIT_TAG 3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96 GIT_SUBMODULES base cmake pluginterfaces public.sdk vstgui4 GIT_SHALLOW FALSE)
  FetchContent_GetProperties(vst3sdk)
  if(NOT vst3sdk_POPULATED)
    FetchContent_Populate(vst3sdk)
  endif()
  set(VST3_SDK_ROOT "${vst3sdk_SOURCE_DIR}")
endif()
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT ON CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR ON CACHE BOOL "" FORCE)
set(VSTGUI_STANDALONE OFF CACHE BOOL "" FORCE)
set(VSTGUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(VSTGUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/sdk")
# The SDK sets this in its own directory; the plugin helper needs it here too.
set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
smtg_add_vst3plugin(VaginaPlugin Source/Processor.cpp Source/Controller.cpp Source/Editor.cpp Source/Factory.cpp)
target_include_directories(VaginaPlugin PRIVATE "${VST3_SDK_ROOT}" Source)
target_link_libraries(VaginaPlugin PRIVATE sdk vstgui_support)
if(APPLE)
  smtg_target_set_bundle(VaginaPlugin BUNDLE_IDENTIFIER "com.leterel.vaginaplugin" COMPANY_NAME "Leterel")
endif()
enable_testing()
add_executable(audio_tests tests/audio_tests.cpp Source/Processor.cpp)
target_include_directories(audio_tests PRIVATE "${VST3_SDK_ROOT}" Source)
target_link_libraries(audio_tests PRIVATE sdk)
target_sources(audio_tests PRIVATE "${public_sdk_SOURCE_DIR}/source/vst/hosting/parameterchanges.cpp" "${public_sdk_SOURCE_DIR}/source/common/memorystream.cpp")
add_test(NAME audio_tests COMMAND audio_tests)
add_executable(meter_lifecycle tests/meter_lifecycle.cpp Source/Processor.cpp Source/Controller.cpp Source/Editor.cpp Source/Factory.cpp)
target_include_directories(meter_lifecycle PRIVATE "${VST3_SDK_ROOT}" Source)
target_link_libraries(meter_lifecycle PRIVATE sdk sdk_hosting vstgui_support)
add_test(NAME meter_lifecycle COMMAND meter_lifecycle)
if(WIN32)
  target_link_libraries(meter_lifecycle PRIVATE ole32)
  add_executable(editor_smoke tests/editor_smoke.cpp)
  target_include_directories(editor_smoke PRIVATE "${VST3_SDK_ROOT}" Source)
  target_link_libraries(editor_smoke PRIVATE sdk sdk_hosting ole32)
  add_dependencies(editor_smoke VaginaPlugin)
  add_test(NAME editor_smoke COMMAND editor_smoke "$<TARGET_FILE:VaginaPlugin>")
  add_executable(visual_render tests/visual_render.cpp Source/Controller.cpp Source/Editor.cpp)
  target_include_directories(visual_render PRIVATE "${VST3_SDK_ROOT}" Source)
  target_link_libraries(visual_render PRIVATE sdk vstgui_support ole32)
  add_test(NAME visual_render COMMAND visual_render)
endif()
